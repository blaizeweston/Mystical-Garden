func verify_full_campaign(nurture: bool) -> void:
	game.leave_game("Campaign fixture reset")
	game.progress.campaign = Campaign.new()
	game.progress.ranged_pressure = 0
	game.progress.parts[0] += 12
	game.destination = Campaign.GROVE
	check(not game.progress.finale_available(), "finale starts sealed")
	for site in range(Campaign.GROVE, Campaign.HEART):
		place(1, Stage.GATE)
		for attempt in Campaign.NAMES.size():
			if game.destination == site:
				break
			game.send_garden_action("destination")
		check(game.destination == site, "successful extraction unlocks the next biome")
		interact()
		for room in range(1, game.ROUTE_ROOMS + 1):
			check(game.depth == room, "campaign enters every room in sequence")
			if room == game.ROUTE_ROOMS:
				check(game.enemy_name() == Campaign.GUARDIANS[site], "route ends at a biome guardian")
			game.scouts.clear()
			game.scouts.clear()
			game.enemy_hp = 1
			game.players[1].strike_at = 0
			place(1, game.enemy_pos)
			game.send_action(true)
			game.update_projectiles(Time.get_ticks_msec()+1000)
			place(1, Stage.GATE if room == game.ROUTE_ROOMS else Stage.DEEPER_GATE)
			interact()
		check(game.progress.campaign.cleared[site]
			and game.progress.campaign.offshoots[site] == Campaign.PENDING,
			"guardian clear plus extraction unlocks an offshoot")
		place(1, Stage.HEART)
		var barbs: int = game.progress.heartwood_barbs
		var grow := nurture and site < Campaign.HOLLOW
		game.send_garden_action("nurture" if grow else "harvest_offshoot")
		check(game.progress.campaign.offshoots[site] == (Campaign.GROWING if grow else Campaign.HARVESTED),
			"offshoot decision is recorded")
		if not grow:
			check(game.progress.heartwood_barbs == barbs + 2, "sacrifice grants immediate crafting power")
	check(not game.progress.finale_available() and game.progress.campaign.memories == [1, 1, 1],
		"one clear of each biome does not open the finale")
	place(1, Stage.GATE)
	for attempt in 6:
		game.send_garden_action("destination")
		check(game.destination != Campaign.HEART, "finale is never a gate route")
	place(1, Stage.HEART)
	game.send_garden_action("awaken")
	check(not game.players[1].wild, "an unrestored Heart cannot be awakened")
	restore_heart_fixture()
	game.progress.knowledge = Campaign.FULL_KNOWLEDGE - 1
	check(not game.progress.finale_available(), "complete memories cannot substitute for knowledge")
	game.progress.knowledge = Campaign.FULL_KNOWLEDGE
	game.progress.rift_hook_discovered = false
	check(not game.progress.finale_available(), "missing research still seals the finale")
	game.progress.rift_hook_discovered = true
	check(game.progress.finale_available(), "completed knowledge and memories unlock the finale")
	game.send_garden_action("awaken")
	check(game.destination == Campaign.HEART and game.enemy_hp == Campaign.FINAL_HP
		and game.enemy_name() == Campaign.ENEMIES[Campaign.HEART], "Heart fight starts")
	game.enemy_hp = 8
	game.attack_pattern = 1
	game.enemy_ready_at = 0
	place(1, game.enemy_pos + Vector3.RIGHT)
	game.update_enemy(0.01, Time.get_ticks_msec())
	check(game.enemy_winding_up and is_equal_approx(game.strike_radius(), 3.65),
		"second phase telegraphs the wider strike")
	game.scouts.clear()
	game.enemy_hp = 1
	game.players[1].strike_at = 0
	game.send_action(true)
	game.update_projectiles(Time.get_ticks_msec()+1000)
	check(game.players[1].wild and game.pending_finale_at>0,"finale allows the death animation before returning")
	var finale_deadline:=Time.get_ticks_msec()+4000
	while game.pending_finale_at>0 and Time.get_ticks_msec()<finale_deadline: await process_frame
	var expected := Campaign.SEED if nurture else Campaign.CORRUPTION
	check(not game.players[1].wild and game.progress.campaign.ending == expected,
		"Heart defeat resolves the ending from mature offshoots (wild=%s ending=%d expected=%d pending=%d now=%d)"%[str(game.players[1].wild),game.progress.campaign.ending,expected,game.pending_finale_at,Time.get_ticks_msec()])
	var reload := Progress.new()
	reload.path = game.progress.path
	check(reload.load_garden() == OK and reload.campaign.ending == expected
		and reload.campaign.offshoots == game.progress.campaign.offshoots,
		"campaign ending and offshoot choices survive save reload")
