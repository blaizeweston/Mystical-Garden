extends SceneTree

const World := preload("res://world.tscn")
const Stage := preload("res://stage.gd")
const Campaign := preload("res://campaign.gd")

var game: Node3D


func _initialize() -> void:
	call_deferred("run")


func run() -> void:
	game = World.instantiate()
	game.progress.path = "res://.godot/campaign_combat_%s.cfg" % Crypto.new().generate_random_bytes(8).hex_encode()
	game.local_character.path = game.progress.path + ".character"
	root.add_child(game)
	var player: Dictionary = game.players[1]
	player.pos = Stage.HEART
	game.send_garden_action("frame")
	var grove_only := OS.get_cmdline_user_args().has("grove")
	var encounter_only := OS.get_cmdline_user_args().has("encounter")
	var last := Campaign.GROVE if grove_only or encounter_only else Campaign.HOLLOW
	game.map_rng.seed = 42016
	player.pos = Stage.GATE
	game.send_action(false)
	if game.loadout_menu.visible: game.send_garden_action("depart")
	if game.checkpoint_pending: game.send_garden_action("checkpoint_choice_-1")
	game.send_garden_action("depart")
	for site in range(last + 1):
		if game.destination != site or not player.wild:
			fail("could not start " + Campaign.NAMES[site])
			return
		for local_encounter in range(1, game.ROUTE_ROOMS + 1):
			if not await fight(player):
				fail("%s encounter %d" % [Campaign.NAMES[site], local_encounter])
				return
			player.pos = game.drop_pos
			game.send_action(false)
			if game.loadout_menu.visible: game.send_garden_action("depart")
			if game.checkpoint_pending: game.send_garden_action("checkpoint_choice_-1")
			if encounter_only:
				player.pos = game.battlefield.advance
				game.send_garden_action("extract")
				if not player.wild:
					fail("ordinary encounter incorrectly allowed extraction")
					return
				print("PASS: exploration encounter combat")
				quit(0)
				return
			if game.encounter_number % game.CHECKPOINT_INTERVAL == 0:
				player.pos = game.battlefield.advance
				game.send_garden_action("upgrade_power")
			player.pos = game.battlefield.advance
			game.send_action(false)
			if game.loadout_menu.visible: game.send_garden_action("depart")
			if game.checkpoint_pending: game.send_garden_action("checkpoint_choice_-1")
			if game.checkpoint_pending: game.send_garden_action("checkpoint_choice_-1")
		print("CLEARED: ", Campaign.NAMES[site])
	if not await reach_extraction(player):
		fail("could not reach a cleared checkpoint")
		return
	if grove_only:
		if not game.progress.campaign.cleared[Campaign.GROVE]:
			fail("Grove clear did not bank")
			return
		print("PASS: Grove combat")
		quit(0)
		return
	for site in Campaign.HEART:
		player.pos = Stage.HEART
		game.send_garden_action("nurture" if site < Campaign.HOLLOW else "harvest_offshoot")
	player.pos = Stage.GATE
	game.send_action(false)
	if game.loadout_menu.visible: game.send_garden_action("depart")
	if game.checkpoint_pending: game.send_garden_action("checkpoint_choice_-1")
	if not await fight(player):
		fail("offshoot growth expedition")
		return
	
	if not await reach_extraction(player):
		fail("offshoot growth checkpoint")
		return
	if game.progress.finale_available():
		fail("first circuit should leave the Heart dormant")
		return
	# This combat check begins the finale from a completed restoration fixture.
	game.progress.flowering_unlocked = true
	game.progress.veilcap_discovered = true
	game.progress.rift_hook_discovered = true
	game.progress.knowledge = Campaign.FULL_KNOWLEDGE
	game.progress.campaign.memories.assign(Campaign.MEMORY_REQUIREMENTS)
	player.pos = Stage.HEART
	game.send_garden_action("awaken")
	if game.destination != Campaign.HEART or not await fight(player):
		fail("Heart finale")
		return
	print("CLEARED: ", Campaign.NAMES[Campaign.HEART])
	if game.progress.campaign.ending != Campaign.SEED:
		fail("expected viable-seed ending")
		return
	print("PASS: campaign combat")
	quit(0)


func fight(player: Dictionary) -> bool:
	var expires := Time.get_ticks_msec() + 60000
	while player.wild and not game.encounter_cleared() and Time.get_ticks_msec() < expires:
		await physics_frame
		if not player.wild or game.encounter_cleared():
			break
		var target: int = game.closest_enemy(player.pos)
		var target_position: Vector3 = game.enemy_position(target)
		var distance: float = target_position.distance_to(player.pos)
		var exposed: bool = game.battlefield.clear_line(player.pos, target_position)
		var hazards: Array = []
		if game.enemy_winding_up and game.enemy_hp > 0:
			hazards.append([game.enemy_landing if game.enemy_attack_mode == "vault" else game.enemy_pos, game.strike_radius()])
		for scout in game.scouts:
			if scout.hp > 0 and scout.windup > 0 and game.Enemies.get_kind(int(scout.get("kind", 0))).role != "chant":
				hazards.append([scout.blast, game.scout_radius(scout)])
		if clearance(player.pos, hazards) < 0.3:
			var escape := best_escape(player.pos, hazards)
			if Time.get_ticks_msec() >= int(player.dodge_at):
				game.dodge_player(1, input_for(escape))
			else:
				game.move_player(1, input_for(escape), 1.0 / 60.0)
		elif not exposed or distance > 3.9:
			var step: Vector3 = game.battlefield.navigate(player.pos, target_position, 0.12, 999)
			var direction: Vector3 = (step - player.pos).normalized()
			game.move_player(1, input_for(direction), 1.0 / 60.0)
		elif distance < 3.1 and target == -1:
			var escape := best_escape(player.pos, [[target_position, 2.8]])
			game.move_player(1, input_for(escape), 1.0 / 60.0)
		var visible: int = game.closest_enemy(player.pos, true)
		if visible != -2 and game.enemy_position(visible).distance_to(player.pos) <= 4.0:
			game.send_garden_action("class_skill")
			game.send_action(true)
	return game.encounter_cleared() or game.progress.campaign.ending != Campaign.UNFINISHED


func clearance(position: Vector3, hazards: Array) -> float:
	var margin := INF
	for hazard in hazards:
		margin = minf(margin, position.distance_to(hazard[0]) - float(hazard[1]))
	return margin


func best_escape(position: Vector3, hazards: Array) -> Vector3:
	var direction := Vector3.RIGHT
	var best := -INF
	for i in 16:
		var candidate := Vector3.RIGHT.rotated(Vector3.UP, TAU * i / 16.0)
		var destination: Vector3 = game.bound_position(position + candidate * 2.8, true)
		var finish: Vector3 = game.battlefield.dash(position, destination)
		var margin := clearance(finish, hazards)
		if margin > best:
			best = margin
			direction = candidate
	return direction


func input_for(direction: Vector3) -> Vector2:
	var input := direction.rotated(Vector3.UP, -atan2(3.3, 14.0))
	return Vector2(input.x, input.z)


func fail(where: String) -> void:
	printerr("FAIL campaign combat: ", where)
	if game != null and game.players.has(1):
		var player: Dictionary = game.players[1]
		printerr("Health ", player.hp, " position ", player.pos, " lead health ", game.enemy_hp)
		for scout in game.scouts:
			if scout.hp > 0:
				printerr("Remaining foe ", scout.pos, " health ", scout.hp)
	quit(1)


func reach_extraction(player: Dictionary) -> bool:
	for step in 6:
		if not game.encounter_cleared():
			if not await fight(player):
				return false
			player.pos = game.drop_pos
			game.send_action(false)
			if game.loadout_menu.visible: game.send_garden_action("depart")
			if game.checkpoint_pending: game.send_garden_action("checkpoint_choice_-1")
		player.pos = game.battlefield.advance
		if game.upgrade_checkpoint_open():
			game.send_garden_action("extract")
			return not player.wild
		game.send_action(false)
		if game.loadout_menu.visible: game.send_garden_action("depart")
		if game.checkpoint_pending: game.send_garden_action("checkpoint_choice_-1")
	return false
